# requirements-macos.txt — development install for the TissueLab AI Service.

# Scope: this service reads pixels, runs the workflow runtime and serves the API.
# It does NOT run model inference. Every tasknode is spawned into its own conda
# env ({model}_tissuelab_ai_service_tasknode) and
# app/utils/workflow/register.py:_isolated_child_env() strips PYTHONPATH and sets
# PYTHONNOUSERSITE=1 specifically so this env's packages cannot leak into it.

# the service source imports none of torch / open-clip / clip-interrogator /
# csbdeep / instanseg / datasets / fairscale / einops, nor scikit-image /
# shapely / rasterio / slideio / xarray / dask / s3fs / numba.
#
# Versions resolved 2026-08-28 for Python 3.11; the full app import graph was
# verified against them. Keep in sync with requirements-packaging.txt.
